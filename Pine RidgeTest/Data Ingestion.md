# load_corpus.py

```python
from pathlib import Path
import json

def load_corpus(path=None):
    if path is None:
        path = Path(__file__).resolve().parents[1] / "synthetic-data" / "pine-ridge-corpus.jsonl"
    records = []
    with Path(path).open("r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                records.append(json.loads(line))
    return records

if __name__ == "__main__":
    records = load_corpus()
    print(f"Loaded {len(records)} chunks")
    print(f"Sources: {sorted(set(r['source_id'] for r in records))}")
```

# test_ingested_corpus.py

```python
from pathlib import Path
import json

ROOT = Path(__file__).resolve().parents[1]
CORPUS = ROOT / "synthetic-data" / "pine-ridge-corpus.jsonl"
MANIFEST = ROOT / "config" / "source-manifest.json"

REQUIRED = {
    "source_id", "source_title", "source_type", "project_id",
    "legal_entity_id", "department", "reporting_period",
    "version_or_effective_date", "approval_status",
    "confidentiality_class", "section_or_page"
}

def read_records():
    return [json.loads(x) for x in CORPUS.read_text(encoding="utf-8").splitlines() if x.strip()]

def test_every_chunk_has_required_metadata():
    records = read_records()
    assert records
    for record in records:
        assert REQUIRED.issubset(record.keys())
        assert record["source_id"]
        assert record["source_title"]
        assert record["source_type"]
        assert record["project_id"]
        assert record["department"]
        assert record["approval_status"] == "Approved"
        assert record["confidentiality_class"] == "Approved for AI"
        assert record["section_or_page"]
        assert record["content"].strip()

def test_chunk_ids_are_unique():
    records = read_records()
    ids = [r["chunk_id"] for r in records]
    assert len(ids) == len(set(ids))

def test_only_approved_reference_sources_are_ingested():
    records = read_records()
    allowed = {"SRC-CASE-001", "SRC-CUR-001", "SRC-SPEC-001"}
    assert set(r["source_id"] for r in records) == allowed

def test_evaluation_suite_is_not_in_retrieval_corpus():
    manifest = json.loads(MANIFEST.read_text(encoding="utf-8"))
    assert "EVALUATION-SUITE(1).md" in manifest["excluded_from_retrieval_corpus"]
    for r in read_records():
        assert "Evaluation Suite" not in r["source_title"]

def test_case_study_is_project_scoped():
    for r in read_records():
        if r["source_id"] == "SRC-CASE-001":
            assert r["project_id"] == "SRI-2026-MF-001"

def test_global_reference_material_is_global_scoped():
    for r in read_records():
        if r["source_id"] in {"SRC-CUR-001", "SRC-SPEC-001"}:
            assert r["project_id"] == "GLOBAL"
```
